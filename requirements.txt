# TripWeaver — Phase 1 (CrewAI orchestration only)
# Install with:  uv pip install -r requirements.txt   (or: pip install -r requirements.txt)
#
# The [anthropic] extra is REQUIRED: it pulls the official `anthropic` SDK so
# CrewAI's native Anthropic provider can talk to Claude. Without it, using an
# `anthropic/...` model raises "Anthropic native provider not available".
crewai[anthropic]>=1.14
python-dotenv>=1.0

# Phase 2 — A2A supplier agents (official A2A SDK).
# Pinned <1.0 on purpose: a2a-sdk 1.x is a protobuf/gRPC-centric rewrite; the
# mature 0.3.x line matches the official A2A Python docs and keeps the example
# clear. (Independent of CrewAI's own a2a wrapper, which we don't use here.)
a2a-sdk>=0.3,<1.0
uvicorn>=0.30   # serves the supplier ASGI apps (and the Phase 3 AG-UI server)
httpx>=0.27     # A2A client transport (and the Phase 3 test client)

# Phase 3 — AG-UI server (stream live agent events to a frontend over SSE).
ag-ui-protocol>=0.1   # AG-UI core event types + SSE EventEncoder
fastapi>=0.115        # the AG-UI HTTP/SSE endpoint

# Phase 4 — A2UI (declarative UI). No new framework needed: the A2UI v0.9 models,
# catalog, and validator ship inside crewai (crewai.a2a.extensions.a2ui). We only
# rely on jsonschema (already pulled in by crewai) to validate emitted surfaces.
jsonschema>=4         # validate A2UI v0.9 messages against the bundled spec schema
